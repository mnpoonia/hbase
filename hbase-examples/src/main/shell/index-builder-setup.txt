# Set up sample data for the IndexBuilder example.
# Run with:  bin/hbase newshell hbase-examples/src/main/shell/index-builder-setup.txt
#       or:  bin/hbase shell -n hbase-examples/src/main/shell/index-builder-setup.txt
create 'people', 'attributes'
create 'people-email', 'INDEX'
create 'people-phone', 'INDEX'
create 'people-name', 'INDEX'

put 'people', '1', 'attributes:name', 'jenny'
put 'people', '1', 'attributes:email', 'jenny@example.com'
put 'people', '1', 'attributes:phone', '867-5309'

put 'people', '2', 'attributes:name', 'alice'
put 'people', '2', 'attributes:email', 'alice@example.com'
put 'people', '2', 'attributes:phone', '555-1234'

put 'people', '3', 'attributes:name', 'kevin'
put 'people', '3', 'attributes:email', 'kevinpet@example.com'
put 'people', '3', 'attributes:phone', '555-1212'
